cmake_minimum_required(VERSION 3.16)
project(gladius_vk_map_probe C)
if(NOT DEFINED ORBIS_COMPAT_DIR OR NOT PS4)
  message(FATAL_ERROR "configure with the Orbis toolchain")
endif()
add_executable(vkmapprobe vk_map_probe.c)
target_compile_options(vkmapprobe PRIVATE -Wall -Wextra -Werror)
add_subdirectory("${ORBIS_KIT_DIR}/vkloader" ps4-vkloader)
target_link_libraries(vkmapprobe PRIVATE ps4-vkloader)
orbis_compat_verify()
if(COMMAND ps4_create_eboot)
  ps4_create_eboot(vkmapprobe)
endif()
