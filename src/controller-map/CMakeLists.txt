cmake_minimum_required(VERSION 3.16)
project(gladius_controller_map C)
if(NOT DEFINED ORBIS_COMPAT_DIR OR NOT PS4)
  message(FATAL_ERROR "configure with -DCMAKE_TOOLCHAIN_FILE=<bundle>/toolchain/orbis-sdk.cmake")
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native" pw_native)
add_executable(controller-map main.c)
target_link_libraries(controller-map PRIVATE pw_native_ps4)
target_compile_options(controller-map PRIVATE -std=c11 -O2 -Wall -Wextra)
if(COMMAND ps4_create_eboot)
  ps4_create_eboot(controller-map)
endif()
