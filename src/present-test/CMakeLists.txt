cmake_minimum_required(VERSION 3.16)
project(pw_present C)
if(NOT DEFINED ORBIS_COMPAT_DIR OR NOT PS4)
  message(FATAL_ERROR "configure with -DCMAKE_TOOLCHAIN_FILE=<bundle>/toolchain/orbis-sdk.cmake")
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native" pw_native)
add_executable(present main.c)
target_link_libraries(present PRIVATE pw_native_ps4)
target_compile_options(present PRIVATE -std=c11 -O2 -Wall -Wextra)
if(COMMAND ps4_create_eboot)
  ps4_create_eboot(present)
endif()
